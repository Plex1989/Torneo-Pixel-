async function abrirScoreboardJuez(partidoId) { if (!partidoId || partidoId === 0) { alert("⚠️ La fase final debe guardarse primero desde 'Sorteo y Rol' para activar el arbitraje oficial de esta llave."); return; } const p = todosLosPartidos.find(x => x.id === partidoId); if (p && !p.grupo_nombre && p.ronda_fase) { const catNombre = p.categorias?.nombre; const partidosCat = todosLosPartidos.filter(m => m.categorias?.nombre === catNombre); const partidosCatGpo = partidosCat.filter(m => m.grupo_nombre && m.grupo_nombre !== 'Flash'); const totalGpo = partidosCatGpo.length; const finGpo = partidosCatGpo.filter(m => String(m.estado || '').trim().toLowerCase() === 'finalizado').length; if (totalGpo > 0 && totalGpo !== finGpo) { alert(`⛔ ACCESO DENEGADO.\n\nNo puedes arbitrar las llaves eliminatorias de la categoría "${catNombre || ''}" porque aún hay partidos de grupo Pendientes o En Vivo.\n\nFinalízalos todos primero.`); return; } const esFinal = String(p.ronda_fase || '').toLowerCase().includes('final') && !String(p.ronda_fase || '').toLowerCase().includes('semi'); if (esFinal) { const semis = partidosCat.filter(m => !m.grupo_nombre && m.ronda_fase && String(m.ronda_fase || '').toLowerCase().includes('semi')); const semisIncompletas = semis.some(m => String(m.estado || '').trim().toLowerCase() !== 'finalizado'); if (semisIncompletas) { alert(`⛔ ACCESO DENEGADO.\n\nNo puedes arbitrar la Gran Final porque las Semifinales aún no han terminado.`); return; } } if (!p.jugador1_id || !p.jugador2_id) { const standings = calcularStandingsGrupos(); const gruposCat = standings[catNombre] || {}; let clasificados = []; Object.keys(gruposCat).forEach(gpoNombre => { let jugArr = ordenarJugadoresITTF(Object.values(gruposCat[gpoNombre])); if(jugArr[0]) clasificados.push({ ...jugArr[0], seedType: 1 }); if(jugArr[1]) clasificados.push({ ...jugArr[1], seedType: 2 }); }); clasificados.sort((a,b) => { if(a.seedType !== b.seedType) return a.seedType - b.seedType; if(b.pts !== a.pts) return b.pts - a.pts; let diffA = (a.setsG||0) - (a.setsP||0); let diffB = (b.setsG||0) - (b.setsP||0); return diffB - diffA; }); let s1 = clasificados[0], s2 = clasificados[1], s3 = clasificados[2], s4 = clasificados[3]; let s5 = clasificados[4], s6 = clasificados[5], s7 = clasificados[6], s8 = clasificados[7]; let resJ1 = null, resJ2 = null; const rf = String(p.ronda_fase || '').toLowerCase(); // Mapeo automático Top 8 Cuartos if (rf.includes('cuartos') && rf.includes('1')) { resJ1 = s1?.id; resJ2 = s8?.id; } else if (rf.includes('cuartos') && rf.includes('2')) { resJ1 = s4?.id; resJ2 = s5?.id; } else if (rf.includes('cuartos') && rf.includes('3')) { resJ1 = s3?.id; resJ2 = s6?.id; } else if (rf.includes('cuartos') && rf.includes('4')) { resJ1 = s2?.id; resJ2 = s7?.id; } // Mapeo Semis else if (rf.includes('semi') && (rf.includes('1') || rf.includes('cruce 1'))) { const c1 = partidosCat.find(m => String(m.ronda_fase || '').toLowerCase().includes('cuartos') && String(m.ronda_fase || '').includes('1')); const c2 = partidosCat.find(m => String(m.ronda_fase || '').toLowerCase().includes('cuartos') && String(m.ronda_fase || '').includes('2')); if(c1 || c2) { resJ1 = c1?.ganador_id; resJ2 = c2?.ganador_id; } else { resJ1 = s1?.id; resJ2 = s4?.id; } } else if (rf.includes('semi') && (rf.includes('2') || rf.includes('cruce 2'))) { const c3 = partidosCat.find(m => String(m.ronda_fase || '').toLowerCase().includes('cuartos') && String(m.ronda_fase || '').includes('3')); const c4 = partidosCat.find(m => String(m.ronda_fase || '').toLowerCase().includes('cuartos') && String(m.ronda_fase || '').includes('4')); if(c3 || c4) { resJ1 = c3?.ganador_id; resJ2 = c4?.ganador_id; } else { resJ1 = s2?.id; resJ2 = s3?.id; } } else if (esFinal) { const semi1 = partidosCat.find(m => String(m.ronda_fase || '').toLowerCase().includes('semi') && (String(m.ronda_fase || '').includes('1') || String(m.ronda_fase || '').includes('cruce 1'))); const semi2 = partidosCat.find(m => String(m.ronda_fase || '').toLowerCase().includes('semi') && (String(m.ronda_fase || '').includes('2') || String(m.ronda_fase || '').includes('cruce 2'))); resJ1 = semi1?.ganador_id; resJ2 = semi2?.ganador_id; } if (resJ1 || resJ2) { let updates = {}; if (resJ1 && !p.jugador1_id) updates.jugador1_id = resJ1; if (resJ2 && !p.jugador2_id) updates.jugador2_id = resJ2; if (Object.keys(updates).length > 0) { await clienteSupabase.from('partidos').update(updates).eq('id', p.id).eq('organizador_id', orgId); } } else { alert("⚠️ No se pudieron determinar los jugadores para este partido. Asegúrate de que las fases anteriores tengan resultados válidos en el sistema."); return; } } } let pin = prompt("🔒 Ingrese el PIN de Autorización para Arbitrar:"); if (pin === pinJuecesActual) { window.location.href = `juez.html?partido=${partidoId}`; } else if (pin !== null) { alert("❌ PIN de autorización incorrecto."); } }
